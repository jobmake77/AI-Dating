```typescript
// app/sitemap.ts
import { createClient } from '@/lib/supabase/server'

export default async function sitemap() {
  const supabase = createClient()

  // 获取所有公开内容
  const { data: posts } = await supabase
    .from('contents')
    .select('id, updated_at')
    .eq('status', 'published')

  // 获取所有用户
  const { data: users } = await supabase
    .from('users')
    .select('username, updated_at')

  const baseUrl = 'https://your-domain.com'

  return [
    {
      url: baseUrl,
      lastModified: new Date(),
      changeFrequency: 'daily',
      priority: 1,
    },
    // 内容页面
    ...(posts?.map((post) => ({
      url: `${baseUrl}/post/${post.id}`,
      lastModified: new Date(post.updated_at),
      changeFrequency: 'weekly',
      priority: 0.8,
    })) || []),
    // 用户主页
    ...(users?.map((user) => ({
      url: `${baseUrl}/u/${user.username}`,
      lastModified: new Date(user.updated_at),
      changeFrequency: 'weekly',
      priority: 0.6,
    })) || []),
  ]
}
```

```typescript
export function OnboardingTour() {
  const [run, setRun] = useState(false)

  useEffect(() => {
    // 检查用户是否完成引导
    const hasCompletedOnboarding = localStorage.getItem('onboarding_completed')
    if (!hasCompletedOnboarding) {
      setRun(true)
    }
  }, [])

  const handleJoyrideCallback = (data) => {
    const { status } = data
    if (status === 'finished' || status === 'skipped') {
      localStorage.setItem('onboarding_completed', 'true')
      // 也可以保存到数据库
      // await updateUserOnboarding(userId, { completed: true })
    }
  }

  return (
    <Joyride
      steps={steps}
      run={run}
      continuous
      showProgress
      showSkipButton
      callback={handleJoyrideCallback}
      styles={{
        options: {
          primaryColor: '#3b82f6',
          zIndex: 10000,
        },
      }}
    />
  )
}
```

```typescript
export function OnboardingProgressCard({ progress }) {

    { id: 'profile', label: '完善个人资料', completed: progress.completed_profile },
    { id: 'post', label: '发布第一篇内容', completed: progress.first_post_published },

  return (

        </div>
        <div className="w-full bg-gray-200 rounded-full h-2">
          <div
            className="bg-blue-600 h-2 rounded-full transition-all"
            style={{ width: `${progressPercent}%` }}
          />
        </div>
      </div>
      <ul className="space-y-2">
        {tasks.map((task) => (
          <li key={task.id} className="flex items-center gap-2">

              <CheckCircle className="w-5 h-5 text-green-500" />

              <Circle className="w-5 h-5 text-gray-300" />
            )}

              {task.label}
            </span>
          </li>
        ))}
      </ul>
    </div>
  )
}
```

```markdown
## 错误处理

| 状态码 | 说明 | 解决方案 |
|--------|------|----------|
| 401 | API Key 无效或过期 | 检查 API Key 是否正确，或重新生成 |
| 429 | 请求过于频繁 | 降低请求频率，遵守速率限制 |
| 500 | 服务器错误 | 稍后重试，或联系支持 |

## 速率限制
- 每个 API Key 每分钟最多 60 次请求
- 每天最多 10,000 次请求
```
